class LayerwiseLRAdam(Adam):
    def __init__(self, special_layers, lr_special=0.0001, lr_normal=0.001, **kwargs):
        super().__init__(learning_rate=lr_normal, **kwargs)
        self.special_layers = special_layers
        self.lr_special = lr_special
        self.lr_normal = lr_normal
    
    def _create_slots(self, var_list):
        for var in var_list:
            self.add_slot(var, 'm') 
            self.add_slot(var, 'v')  
    def _resource_apply_dense(self, grad, var, apply_state=None):
        if not hasattr(self, '_slots'):
            self._slots = {}
        
        var_key = self._var_key(var)
        if var_key not in self._slots:
            self._slots[var_key] = {}
            self.add_slot(var, 'm')
            self.add_slot(var, 'v')
        
        lr_value = self.lr_special if any(layer_name in var.name for layer_name in self.special_layers) else self.lr_normal
        
        if hasattr(lr_value, '__call__'):
            lr_value = lr_value(self.iterations)
        lr = tf.convert_to_tensor(lr_value, dtype=var.dtype)

        beta_1 = self._get_hyper('beta_1', var.dtype)
        beta_2 = self._get_hyper('beta_2', var.dtype)
        

        epsilon = tf.convert_to_tensor(self.epsilon, dtype=var.dtype)
        
        m = self.get_slot(var, 'm')
        v = self.get_slot(var, 'v')
        
        local_step = tf.cast(self.iterations + 1, var.dtype)

        beta_1_power = tf.pow(beta_1, local_step)
        beta_2_power = tf.pow(beta_2, local_step)
        lr_t = lr * tf.sqrt(1 - beta_2_power) / (1 - beta_1_power)

        m_t = beta_1 * m + (1 - beta_1) * grad
        v_t = beta_2 * v + (1 - beta_2) * tf.square(grad)
        var_t = var - lr_t * m_t / (tf.sqrt(v_t) + epsilon)

        return tf.group(*[var.assign(var_t), m.assign(m_t), v.assign(v_t)])
    
    def _resource_apply_sparse(self, grad, var, indices, apply_state=None):
        if not hasattr(self, '_slots'):
            self._slots = {}

        var_key = self._var_key(var)
        if var_key not in self._slots:
            self._slots[var_key] = {}
            self.add_slot(var, 'm')
            self.add_slot(var, 'v')

        lr_value = self.lr_special if any(layer_name in var.name for layer_name in self.special_layers) else self.lr_normal

        if hasattr(lr_value, '__call__'):
            lr_value = lr_value(self.iterations)
        
        lr = tf.convert_to_tensor(lr_value, dtype=var.dtype)

        beta_1 = self._get_hyper('beta_1', var.dtype)
        beta_2 = self._get_hyper('beta_2', var.dtype)

        epsilon = tf.convert_to_tensor(self.epsilon, dtype=var.dtype)
        
        m = self.get_slot(var, 'm')
        v = self.get_slot(var, 'v')

        local_step = tf.cast(self.iterations + 1, var.dtype)

        beta_1_power = tf.pow(beta_1, local_step)
        beta_2_power = tf.pow(beta_2, local_step)
        lr_t = lr * tf.sqrt(1 - beta_2_power) / (1 - beta_1_power)

        m_t = beta_1 * m
        m_t = tf.tensor_scatter_nd_add(m_t, tf.expand_dims(indices, 1), (1 - beta_1) * grad)
        
        v_t = beta_2 * v
        v_t = tf.tensor_scatter_nd_add(v_t, tf.expand_dims(indices, 1), (1 - beta_2) * tf.square(grad))
        
        var_t = var - lr_t * m_t / (tf.sqrt(v_t) + epsilon)

        return tf.group(*[var.assign(var_t), m.assign(m_t), v.assign(v_t)])
    
    def _var_key(self, var):
        if hasattr(var, 'op'):
            return var.op.name
        return var.name
    
    def get_config(self):
        config = super().get_config()
        config.update({
            'special_layers': self.special_layers,
            'lr_special': self.lr_special,
            'lr_normal': self.lr_normal})
        return config

def abs_backend(inputs):
    return K.abs(inputs)

def expand_dim_backend(inputs):
    return K.expand_dims(inputs,1)

def sign_backend(inputs):
    return K.sign(inputs)

def pad_backend(inputs, in_channels, out_channels):
    pad_dim = (out_channels - in_channels)//2
    inputs = K.expand_dims(inputs)
    inputs = K.spatial_2d_padding(inputs,padding=((0,0),(pad_dim,pad_dim)))
    return K.squeeze(inputs,-1)

def residual_shrinkage_block(incoming, nb_blocks, out_channels, downsample=False,
                             downsample_strides=2):
  
  
    residual = incoming
    in_channels = incoming.get_shape().as_list()[-1]
  
    for _ in range(nb_blocks):
      
        identity = residual
      
        if not downsample:
            downsample_strides = 1
      
        residual = BatchNormalization()(residual)
        residual = Activation('relu')(residual)
        residual = Conv1D(out_channels, 3, strides=downsample_strides, 
                          padding='same', kernel_initializer='he_normal', 
                          kernel_regularizer=l2(1e-4))(residual)
      
        residual = BatchNormalization()(residual)
        residual = Activation('relu')(residual)
        residual = Conv1D(out_channels, 3, padding='same', kernel_initializer='he_normal', 
                          kernel_regularizer=l2(1e-4))(residual)
      
        residual_abs = Lambda(abs_backend)(residual)
        abs_mean = GlobalAveragePooling1D()(residual_abs)
      
        scales = Dense(out_channels, activation=None, kernel_initializer='he_normal', 
                       kernel_regularizer=l2(1e-4))(abs_mean)
        scales = BatchNormalization()(scales)
        scales = Activation('relu')(scales)
        scales = Dense(out_channels, activation='sigmoid', kernel_regularizer=l2(1e-4))(scales)
        scales = Lambda(expand_dim_backend)(scales)

        thres = keras.layers.multiply([abs_mean, scales])

        sub = keras.layers.subtract([residual_abs, thres])
        zeros = keras.layers.subtract([sub, sub])
        n_sub = keras.layers.maximum([sub, zeros])
        residual = keras.layers.multiply([Lambda(sign_backend)(residual), n_sub])

        if downsample_strides > 1:
            identity = AveragePooling1D(pool_size=1, strides=2)(identity)

        if in_channels != out_channels:
        
            identity = Lambda(pad_backend, arguments={'in_channels':in_channels,'out_channels':out_channels})(identity)
        
        residual = keras.layers.add([residual, identity])
    return residual



def residual_shrinkage_block2D(incoming, nb_blocks, size, out_channels, downsample=False,
                             downsample_strides=2):
    residual = incoming
    in_channels = incoming.get_shape().as_list()[-1]
    
    for i in range(nb_blocks):
        
        identity = residual
        
        if not downsample: 
            downsample_strides = 1
        residual = BatchNormalization()(residual)
        residual = Activation('relu')(residual)
        residual = Conv2D(out_channels, size, strides=(downsample_strides, downsample_strides), 
                          padding='same', kernel_initializer='he_normal', 
                          kernel_regularizer=l2(1e-4))(residual)
        
        residual = BatchNormalization()(residual)
        residual = Activation('relu')(residual)
        residual = Conv2D(out_channels, size, padding='same', kernel_initializer='he_normal', 
                          kernel_regularizer=l2(1e-4))(residual)
                       
        # Calculate global means
        residual_abs = Lambda(abs_backend)(residual)
        abs_mean = GlobalAveragePooling2D()(residual_abs)
        abs_mean1 = Reshape((out_channels,1))(abs_mean)
        
        # ECA
        kernel_size = int(abs((math.log(out_channels, 2) + 1) /2))
        kernel_size = kernel_size if kernel_size % 2 else kernel_size + 1
        scales = Conv1D(1, kernel_size,activation='sigmoid', padding='same')(abs_mean1)
        scales = Reshape((1, out_channels))(scales)
    
        thres = keras.layers.multiply([abs_mean, scales])
        
        # Soft thresholding
        sub = keras.layers.subtract([residual_abs, thres])
        zeros = keras.layers.subtract([sub, sub])
        n_sub = keras.layers.maximum([sub, zeros])
        residual = keras.layers.multiply([Lambda(sign_backend)(residual), n_sub])
        
        if downsample_strides > 1:
            identity = AveragePooling2D(pool_size=(1, 1), strides=(2, 2))(identity)
            
        if in_channels != out_channels:
            identity = Lambda(pad_backend, arguments={'in_channels':in_channels,'out_channels':out_channels})(identity)
        
        residual = keras.layers.add([residual, identity])
    
    return residual

class AdaptiveWeightMerge(tf.keras.layers.Layer):
    def __init__(self):
        super(AdaptiveWeightMerge, self).__init__()
        self.w = self.add_weight(
            name="adaptive_weight",
            shape=(1,),
            initializer="zeros",
            trainable=True)

    def call(self, inputs):
        prob_a, prob_b = inputs
        alpha = tf.sigmoid(self.w) 
        return alpha * prob_a + (1 - alpha) * prob_b

class PositionalEncoding(layers.Layer):
    def __init__(self, seq_len, embed_dim):
        super(PositionalEncoding, self).__init__()
        self.seq_len = seq_len
        self.embed_dim = embed_dim

    def build(self, input_shape):

        position = tf.range(self.seq_len, dtype=tf.float32)[:, tf.newaxis]
        div_term = tf.range(self.embed_dim, dtype=tf.float32)[tf.newaxis, :]
        div_term = tf.pow(10000.0, -2.0 * (div_term // 2) / tf.cast(self.embed_dim, tf.float32))

        angle_rads = position * div_term

        sines = tf.math.sin(angle_rads[:, 0::2])
        cosines = tf.math.cos(angle_rads[:, 1::2])

        pos_encoding = tf.concat([sines, cosines], axis=-1)
        pos_encoding = pos_encoding[tf.newaxis, ...] 
        self.pos_encoding = tf.cast(pos_encoding, dtype=tf.float32)

    def call(self, inputs):
        return inputs + self.pos_encoding[:, :tf.shape(inputs)[1], :]

class TransformerBlock(layers.Layer):
    def __init__(self, embed_dim, num_heads, ff_dim, rate=0.1):
        super(TransformerBlock, self).__init__()
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim)
        self.ffn = keras.Sequential([
            layers.Dense(ff_dim, activation="gelu", kernel_regularizer=regularizers.l2(1e-4)),
            layers.Dropout(rate),
            layers.Dense(embed_dim, kernel_regularizer=regularizers.l2(1e-4))
        ])
        self.layernorm1 = layers.LayerNormalization(epsilon=1e-6)
        self.layernorm2 = layers.LayerNormalization(epsilon=1e-6)
        self.dropout1 = layers.Dropout(rate)
        self.dropout2 = layers.Dropout(rate)

    def call(self, inputs, training):
        attn_output = self.att(inputs, inputs)
        attn_output = self.dropout1(attn_output, training=training)
        out1 = self.layernorm1(inputs + attn_output)
        ffn_output = self.ffn(out1)
        ffn_output = self.dropout2(ffn_output, training=training)
        return self.layernorm2(out1 + ffn_output)

        if current_epoch not in self.target_epochs:
            return
